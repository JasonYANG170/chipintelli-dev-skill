![1](https://document.chipintelli.com/%E8%BD%AF%E4%BB%B6%E5%BC%80%E5%8F%91/SDK/CI230X%E8%8A%AF%E7%89%87SDK/CI230X-DEMO/%E6%B6%82%E9%B8%A6%E7%BA%A2%E5%A4%96%E9%81%A5%E6%8E%A7%E5%99%A8%E4%BD%BF%E7%94%A8%E8%AF%B4%E6%98%8E/img/%E6%B6%82%E9%B8%A6%E7%BA%A2%E5%A4%96%E9%81%A5%E6%8E%A7%E5%99%A8%E4%BD%BF%E7%94%A8%E8%AF%B4%E6%98%8E-1.png)
